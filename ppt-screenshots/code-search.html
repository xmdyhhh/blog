// 配置: Content API Key 从主题注入
const CONFIG = {
  apiUrl: location.origin + '/ghost/api/content/posts/',
  apiKey: document.documentElement.dataset.contentApiKey,
  searchDebounce: 300,   // 防抖300ms
  maxResults: 10
};

// Ctrl/Cmd + K 快捷键唤起
document.addEventListener('keydown', (e) => {
  if ((e.ctrlKey || e.metaKey) && e.key === 'k') {
    e.preventDefault();
    toggleSearch();
  }
});

// 防抖搜索
input.addEventListener('input', () => {
  clearTimeout(timer);
  timer = setTimeout(() => performSearch(query), 300);
});

// 调 Ghost Content API, 前端只暴露只读Key
async function performSearch(q) {
  const url = CONFIG.apiUrl + '?key=' + CONFIG.apiKey
    + '&filter=' + encodeURIComponent('(title:~'+q+',excerpt:~'+q+')');
  const {posts} = await fetch(url).then(r=>r.json());
  renderResults(posts, q);  // 关键词高亮+键盘导航
}

// 无结果时推荐热门标签
function getNoResults(q) {
  return `未找到"${q}", 试试: #开源软件 #技术教程 #Node.js`;
}
